```csharp
using System;
using Tichu.Net;

namespace Tichu.Presentation.Shell
{
    public enum OnlineEntryKind { AskAddress, StartDirect, Unavailable }

    public readonly struct OnlineEntryDecision
    {
        public readonly OnlineEntryKind Kind;
        public readonly Uri Uri;   // StartDirect 일 때만

        public OnlineEntryDecision(OnlineEntryKind kind, Uri uri)
        {
            Kind = kind;
            Uri = uri;
        }
    }

    /// <summary>
    /// "온라인 대전" 버튼을 눌렀을 때 어디로 갈지(순수, 유니티 무참조). 개발 빌드는 늘 주소 입력 화면,
    /// 출시 빌드는 빌드에 넣은 출시 주소로 바로 접속하고 주소가 없거나 잘못됐으면 '준비 중'.
    /// </summary>
    public static class OnlineEntry
    {
        public static OnlineEntryDecision Decide(bool isDebugBuild, string releaseAddress)
        {
            if (isDebugBuild) return new OnlineEntryDecision(OnlineEntryKind.AskAddress, null);
            return ServerAddress.TryParse(releaseAddress, out var uri)
                ? new OnlineEntryDecision(OnlineEntryKind.StartDirect, uri)
                : new OnlineEntryDecision(OnlineEntryKind.Unavailable, null);
        }

        /// <summary>개발 빌드 주소 입력란 초기값: 기억한 주소 → 출시 주소 → 빈칸.</summary>
        public static string InitialAddress(string saved, string releaseAddress)
        {
            if (!string.IsNullOrWhiteSpace(saved)) return saved;
            return releaseAddress ?? "";
        }
    }
}
```

using System.Collections.Generic;
using Tichu.Net;

namespace Tichu.Presentation.Net
{
    /// <summary>
    /// 서버가 보낸 거부·중단 사유(대부분 영문 디버그 문자열)를 사용자 문구로 바꾼다(순수, 유니티 무참조).
    /// 알려진 폴트 문구는 서버 리터럴을 복제해 둔 것 — 서버 문구가 바뀌면 일반 문구로 떨어질 뿐 깨지지 않는다.
    /// </summary>
    public static class NetMessages
    {
        public const string LateAnswer = "시간이 지나 AI가 대신 처리했습니다";
        public const string CannotDo = "낼 수 없는 동작입니다";
        public const string ServerError = "서버 오류로 판이 중단되었습니다";

        // 클라이언트(RoundBootstrap)가 직접 내는 폴트 사유.
        public const string CannotConnect = "서버에 연결할 수 없습니다";
        public const string ConnectionLost = "서버와 연결이 끊어졌습니다";
        public const string MatchGone = "진행 중인 판이 끝났습니다";

        // 시간 초과로 대리 AI 가 이미 둔 질문을 향한 늦은 답.
        private static readonly HashSet<string> Late = new HashSet<string>
        {
            "stale answer",
            "stale action from a previous round",
        };

        // 재접속 뒤 다시 뜬 질문처럼 내 답이 이미 쓰였을 수 있는 거부 — 알리지 않는다.
        private const string AlreadyDecided = "already decided this round";

        private static readonly HashSet<string> KnownFaults = new HashSet<string>
        {
            "서버 연결 끊김",           // WebSocketClientTransport.LostReason(private)
            "다른 곳에서 접속",         // 서버 Matchmaker/Room
            MatchGone,                  // 서버 Room + 클라 재접속
            CannotConnect,
            ConnectionLost,
            ProtocolVersion.UpdateRequiredReason,
        };

        /// <summary>판 중 거부 → 토스트 문구. null 이면 알리지 않는다.</summary>
        public static string ForRejection(string reason)
        {
            if (reason == AlreadyDecided) return null;
            return reason != null && Late.Contains(reason) ? LateAnswer : CannotDo;
        }

        /// <summary>판 중단 사유 → 메뉴 알림 문구. null(사용자 취소)은 null 그대로.</summary>
        public static string ForFault(string reason)
        {
            if (reason == null) return null;
            return KnownFaults.Contains(reason) ? reason : ServerError;
        }
    }

            {
                if (_reconnecting) return LobbyDisplay.Reconnecting;
                if (Required > 0) return new LobbyDisplay("곧 시작합니다", Math.Min(Connected, 4), "다른 플레이어 기다리는 중");
                if (Busy) return new LobbyDisplay("서버가 붐빕니다", 0, $"대기 순번 {Position} · 자리가 나면 자동으로 들어갑니다");
                if (_fillAt.HasValue)
                {
                    int left = Math.Max(0, (int)Math.Ceiling(_fillAt.Value - _now()));
                    return new LobbyDisplay("상대를 찾는 중", Math.Min(Waiting, 4), $"{left}초 뒤 빈자리는 AI로 채우고 시작합니다");
                }
                return LobbyDisplay.Connecting;
            }
        }

        {
            var v = new LobbyOverlayView();

            v.Root = new GameObject("LobbyOverlay", typeof(RectTransform), typeof(Image));

            row.sizeDelta = new Vector2(4 * 180f + 3 * 24f, 160f);

            {

                seat.transform.SetParent(row, false);
                var srt = (RectTransform)seat.transform;
                srt.anchorMin = srt.anchorMax = srt.pivot = new Vector2(0f, 0.5f);
                srt.sizeDelta = new Vector2(180f, 160f);
                srt.anchoredPosition = new Vector2(i * (180f + 24f), 0f);
                v._seatBg[i] = seat.GetComponent<Image>();
                v._seatLabel[i] = NewText("Label", seat.transform, font, 32, Vector2.zero, Vector2.zero);
                Stretch(v._seatLabel[i].rectTransform);
            }

            v._detail = NewText("Detail", v.Root.transform, font, 30, new Vector2(1400f, 60f), new Vector2(0f, -90f));

            var btn = new GameObject("Btn_취소", typeof(RectTransform), typeof(Image), typeof(Button));
            btn.transform.SetParent(v.Root.transform, false);
            var brt = (RectTransform)btn.transform;
            brt.anchorMin = brt.anchorMax = brt.pivot = new Vector2(0.5f, 0.5f);
            brt.sizeDelta = new Vector2(360f, 96f); brt.anchoredPosition = new Vector2(0f, -200f);
            btn.GetComponent<Image>().color = ButtonColor;
            v.Cancel = btn.GetComponent<Button>();
            var lbl = NewText("L", btn.transform, font, 36, Vector2.zero, Vector2.zero);

            return v;
        }

        public void Render(LobbyDisplay d)
        {
            _title.text = d.Title;
            _detail.text = d.Detail;
            _seatRow.SetActive(d.Seats > 0);
            for (int i = 0; i < 4; i++)
            {
                bool filled = i < d.Seats;
                _seatBg[i].color = filled ? SeatFilled : SeatEmpty;
                _seatLabel[i].text = i == 0 ? "나" : filled ? "플레이어" : "빈자리";
            }
        }

        public void Destroy() => Object.Destroy(Root);

        private static Text NewText(string name, Transform parent, Font font, int size, Vector2 box, Vector2 pos)
        {
            var go = new GameObject(name, typeof(RectTransform), typeof(Text));
            go.transform.SetParent(parent, false);
            var t = go.GetComponent<Text>();
            t.font = font; t.fontSize = size; t.alignment = TextAnchor.MiddleCenter; t.color = Ink;
            var rt = t.rectTransform;
            rt.anchorMin = rt.anchorMax = rt.pivot = new Vector2(0.5f, 0.5f);
            rt.sizeDelta = box; rt.anchoredPosition = pos;
            return t;
        }

        private static void Stretch(RectTransform rt)
        {
            rt.anchorMin = Vector2.zero; rt.anchorMax = Vector2.one;
            rt.offsetMin = Vector2.zero; rt.offsetMax = Vector2.zero;
        }
    }
